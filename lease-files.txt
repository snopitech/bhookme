const API_BASE_URL = import.meta.env.VITE_API_URL
  ? `${import.meta.env.VITE_API_URL}/api`
  : 'http://localhost:8080/api';

const MAX_FILE_BYTES = 10 * 1024 * 1024; // matches the backend's 10MB limit
// JPEG/PNG/WebP only: on iPhones this makes the photo picker hand over
// JPEGs instead of HEIC files (which the server can't resize and most
// browsers can't display).
const ACCEPT = 'image/jpeg,image/png,image/webp';
const PARALLEL_UPLOADS = 3;

let nextId = 1;

// Uploads one file to /api/upload/image with progress reporting.
function uploadFile(file, folder, token, onProgress) {
  return new Promise((resolve, reject) => {
    const form = new FormData();
    form.append('file', file);
    if (folder) form.append('folder', folder);

    const xhr = new XMLHttpRequest();
    xhr.open('POST', `${API_BASE_URL}/upload/image`);
    xhr.setRequestHeader('Authorization', `Bearer ${token}`);
    xhr.upload.onprogress = (e) => {
      if (e.lengthComputable) onProgress(Math.round((e.loaded / e.total) * 100));
    };
    xhr.onload = () => {
      let data = null;
      try {
        data = JSON.parse(xhr.responseText);
      } catch {
        // non-JSON error body
      }
      if (xhr.status >= 200 && xhr.status < 300 && data?.url) resolve(data.url);
      else reject(new Error(data?.message || `Upload failed (status ${xhr.status}).`));
    };
    xhr.onerror = () => reject(new Error('Could not reach the server.'));
    xhr.send(form);
  });
}

/**
 * Multi-photo uploader: pick or drop many photos, they upload in the background
 * (3 at a time, resized/compressed by the server), then can be reordered or removed.
 * The first photo is the cover.
 *
 *   <PhotoUploader value={urls} onChange={setUrls} folder="properties" max={30} />
 *
 * `value` is the list of uploaded photo URLs. `onUploadingChange(bool)` lets a form
 * disable its submit button while uploads are still running.
 */
export default function PhotoUploader({ value = [], onChange, folder, max = 30, onUploadingChange }) {
  const { token } = useAuth();
  const inputRef = useRef(null);
  const [pending, setPending] = useState([]); // { id, name, preview, progress, error }
  const [dragOver, setDragOver] = useState(false);
  const [dragIndex, setDragIndex] = useState(null);
  const [notice, setNotice] = useState('');
  const urlsRef = useRef(value);

  useEffect(() => {
    urlsRef.current = value;
  }, [value]);

  const uploading = pending.some((p) => !p.error);
  useEffect(() => {
    onUploadingChange?.(uploading);
  }, [uploading, onUploadingChange]);

  // Free the local preview images when they're no longer shown.
  useEffect(
    () => () => {
      pending.forEach((p) => URL.revokeObjectURL(p.preview));
    },
    // eslint-disable-next-line react-hooks/exhaustive-deps
    []
  );

  const updatePending = (id, patch) =>
    setPending((prev) => prev.map((p) => (p.id === id ? { ...p, ...patch } : p)));

  const addFiles = async (fileList) => {
    setNotice('');
    const files = Array.from(fileList || []);
    if (files.length === 0) return;

    const room = max - urlsRef.current.length - pending.filter((p) => !p.error).length;
    if (room <= 0) {
      setNotice(`You can add up to ${max} photos.`);
      return;
    }

    const accepted = [];
    const skipped = [];
    files.forEach((file) => {
      if (!file.type.startsWith('image/')) skipped.push(`${file.name} is not an image`);
      else if (file.size > MAX_FILE_BYTES) skipped.push(`${file.name} is over 10MB`);
      else accepted.push(file);
    });
    if (accepted.length > room) {
      skipped.push(`${accepted.length - room} photo(s) over the ${max}-photo limit`);
      accepted.length = room;
    }
    if (skipped.length) setNotice(`Skipped: ${skipped.join('; ')}.`);

    const jobs = accepted.map((file) => ({
      id: nextId++,
      name: file.name,
      preview: URL.createObjectURL(file),
      progress: 0,
      error: '',
      file,
    }));
    setPending((prev) => [...prev, ...jobs.map(({ file, ...rest }) => rest)]);

    // Upload a few at a time, adding each URL as soon as it finishes.
    let index = 0;
    const worker = async () => {
      while (index < jobs.length) {
        const job = jobs[index++];
        try {
          const url = await uploadFile(job.file, folder, token, (progress) => updatePending(job.id, { progress }));
          const next = [...urlsRef.current, url];
          urlsRef.current = next;
          onChange(next);
          URL.revokeObjectURL(job.preview);
          setPending((prev) => prev.filter((p) => p.id !== job.id));
        } catch (err) {
          updatePending(job.id, { error: err.message, progress: 0 });
        }
      }
    };
    await Promise.all(Array.from({ length: Math.min(PARALLEL_UPLOADS, jobs.length) }, worker));
  };

  const removeUrl = (url) => onChange(value.filter((u) => u !== url));

  const dismissFailed = (id) =>
    setPending((prev) => {
      const job = prev.find((p) => p.id === id);
      if (job) URL.revokeObjectURL(job.preview);
      return prev.filter((p) => p.id !== id);
    });

  const move = (from, to) => {
    if (to < 0 || to >= value.length || from === to) return;
    const next = [...value];
    const [item] = next.splice(from, 1);
    next.splice(to, 0, item);
    onChange(next);
  };

  const total = value.length + pending.filter((p) => !p.error).length;

  return (
    <div>
      {/* Drop zone */}
      <div
        onDragOver={(e) => {
          if (e.dataTransfer.types.includes('Files')) {
            e.preventDefault();
            setDragOver(true);
          }
        }}
        onDragLeave={() => setDragOver(false)}
        onDrop={(e) => {
          if (!e.dataTransfer.files?.length) return;
          e.preventDefault();
          setDragOver(false);
          addFiles(e.dataTransfer.files);
        }}
        onClick={() => inputRef.current?.click()}
        onKeyDown={(e) => (e.key === 'Enter' || e.key === ' ') && inputRef.current?.click()}
        role="button"
        tabIndex={0}
        className={`flex flex-col items-center justify-center text-center border-2 border-dashed rounded-xl px-4 py-8 cursor-pointer transition ${
          dragOver ? 'border-rose-500 bg-rose-50' : 'border-gray-300 hover:border-gray-500 bg-gray-50'
        }`}
      >
        <span className="text-3xl mb-2" aria-hidden="true">📷</span>
        <p className="font-semibold text-sm">Drag photos here or click to choose</p>
        <p className="text-xs text-gray-500 mt-1">
          JPG, PNG or WebP, up to 10MB each, up to {max} photos. Large photos are resized automatically.
        </p>
        <input
          ref={inputRef}
          type="file"
          accept={ACCEPT}
          multiple
          className="hidden"
          onChange={(e) => {
            addFiles(e.target.files);
            e.target.value = '';
          }}
        />
      </div>

      {notice && <p className="text-sm text-amber-700 mt-2">{notice}</p>}

      {(value.length > 0 || pending.length > 0) && (
        <>
          <p className="text-xs text-gray-500 mt-3 mb-2">
            {total} of {max} photos · drag to reorder · the first photo is the cover

                onDragOver={(e) => e.preventDefault()}
                onDrop={(e) => {

                  if (dragIndex !== null) move(dragIndex, i);

  if (!dateString) return '';
  const [y, m, d] = dateString.split('-').map(Number);

function formatSlot(iso) {
  const [date, time] = iso.split('T');
  const [y, m, d] = date.split('-').map(Number);
  const [hh, mm] = time.split(':').map(Number);
  return new Date(y, m - 1, d, hh, mm).toLocaleString('en-US', {
    weekday: 'short',
    month: 'short',
    day: 'numeric',
    hour: 'numeric',
    minute: '2-digit',
  });
}

function todayISO() {
  const now = new Date();
  return new Date(now.getTime() - now.getTimezoneOffset() * 60000).toISOString().slice(0, 10);

async function apiCall(path, token, options = {}) {
  const res = await fetch(`${API_BASE_URL}${path}`, {
    ...options,
    headers: {
      ...(options.body ? { 'Content-Type': 'application/json' } : {}),
      Authorization: `Bearer ${token}`,
    },
  });
  const data = await res.json().catch(() => null);

}

function SlotManager({ token, experience, onChanged }) {

    }
    setBusy(true);
    setError('');

        method: 'POST',
        body: JSON.stringify({ startTime: `${date}T${time}`, capacity: Number(capacity) }),

      setError(err.message);

      setError(err.message);
    } finally {
      setBusy(false);
    }
  };

      <div className="flex flex-wrap items-end gap-2 mb-3">

        <Field label="Spots">
          <input
            type="number"
            min="1"
            max={experience.maxGuests}

  const [sending, setSending] = useState(false);
  const bottomRef = useRef(null);

  const conversationUrl =
    thread.listingType === 'car'
      ? `${API_BASE_URL}/messages/conversation/car/${thread.carId}/with/${thread.partnerId}`
      : `${API_BASE_URL}/messages/conversation/property/${thread.propertyId}/with/${thread.partnerId}`;

  const load = useCallback(
    (quiet) => {
      if (!quiet) setLoading(true);
      return fetch(conversationUrl, { headers: { Authorization: `Bearer ${token}` } })
        .then(async (res) => {
          const data = await res.json().catch(() => null);
          if (!res.ok) throw new Error((data && data.message) || 'Could not load this conversation.');
          setMessages(Array.isArray(data) ? data : []);
          setError('');
        })
        .catch((err) => setError(err.message))
        .finally(() => setLoading(false));
    },
    [conversationUrl, token]
  );

    load(false);
    const timer = setInterval(() => load(true), REFRESH_MS);
    return () => clearInterval(timer);

// Clear any MSAL-managed sessionStorage/localStorage keys. MSAL uses
// keys prefixed with "msal." (e.g. msal.<clientId>.account, msal.token.keys...).
// Without clearing these on logout, MSAL still thinks the user is
// signed in, and the next "Sign in with Microsoft" click silently
// reuses the cached account - skipping the account picker entirely.
function clearMsalCache() {
  try {
    const prefixes = ['msal.'];
    for (const storage of [sessionStorage, localStorage]) {
      const keysToRemove = [];
      for (let i = 0; i < storage.length; i++) {
        const key = storage.key(i);
        if (key && prefixes.some((p) => key.startsWith(p))) {
          keysToRemove.push(key);
        }
      }
      keysToRemove.forEach((k) => storage.removeItem(k));
    }
  } catch (e) {
    console.error('clearMsalCache failed:', e);
  }
}

  };

    }
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [code]);

  const handleChange = (index, value) => {
    if (value && !/^\d+$/.test(value)) return;
    const newCode = [...code];
    newCode[index] = value.slice(0, 1);
    setCode(newCode);
    if (value && index < 5) {
      inputRefs.current[index + 1]?.focus();
    }
  };

  const handleKeyDown = (index, e) => {
    if (e.key === 'Backspace' && !code[index] && index > 0) {
      inputRefs.current[index - 1]?.focus();
    }
  };

  const handlePaste = (e) => {
    e.preventDefault();
    const pasted = e.clipboardData.getData('text').trim();
    if (/^\d{6}$/.test(pasted)) {
      setCode(pasted.split(''));
    }
  };

  const handleResend = async () => {
    if (!canResend) return;
    setResending(true);
    setError('');
    setMessage('');
    try {
      const res = await fetch(`${API_BASE_URL}/auth/host-resend-totp`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ email }),
      });
      const data = await res.json();
      if (res.ok && data.success) {
        setMessage('New verification code sent to your email.');
        setTimeLeft(300);
        setCanResend(false);
        setCode(['', '', '', '', '', '']);
        inputRefs.current[0]?.focus();
      } else {
        setError(data.message || 'Failed to resend code.');
      }
    } catch (err) {
      setError('Network error. Please try again.');
    } finally {
      setResending(false);
    }
  };

  const formatTime = (seconds) => {
    const mins = Math.floor(seconds / 60);
    const secs = seconds % 60;
    return `${mins}:${secs.toString().padStart(2, '0')}`;
  };

  return (
    <div className="max-w-md mx-auto px-6 py-10">
      <div className="text-center mb-6">
        <p className="text-4xl mb-2">🔐</p>
        <h1 className="text-2xl font-bold">Verify your identity</h1>
        <p className="text-sm text-gray-500 mt-1">Enter the 6-digit code sent to your email</p>
        <p className="text-xs text-gray-400 mt-1">{email}</p>
      </div>

      {error && (
        <div className="mb-4 p-3 bg-red-50 border border-red-200 text-red-700 rounded-lg text-sm">{error}</div>
      )}
      {message && (
        <div className="mb-4 p-3 bg-green-50 border border-green-200 text-green-700 rounded-lg text-sm">{message}</div>
      )}

      <div className="grid grid-cols-6 gap-2 mb-6">
        {code.map((digit, index) => (
          <input
            key={index}
            ref={(el) => (inputRefs.current[index] = el)}
            type="text"
            inputMode="numeric"
            maxLength={1}
            value={digit}
            onChange={(e) => handleChange(index, e.target.value)}
            onKeyDown={(e) => handleKeyDown(index, e)}
            onPaste={index === 0 ? handlePaste : undefined}
            disabled={loading}
            autoFocus={index === 0}
            className="w-full aspect-square text-center text-2xl font-bold border-2 border-gray-300 rounded-lg focus:outline-none focus:ring-2 focus:ring-rose-500"
          />
        ))}
      </div>

      <div className="flex items-center justify-between text-sm">
        <span className="text-gray-500">
          Code expires in <span className={timeLeft < 60 ? 'text-red-600 font-semibold' : 'font-semibold'}>{formatTime(timeLeft)}</span>
        </span>
        <button
          onClick={handleResend}
          disabled={!canResend || resending}
          className={`font-medium ${canResend && !resending ? 'text-rose-500 hover:underline' : 'text-gray-400 cursor-not-allowed'}`}
        >
          {resending ? 'Sending...' : 'Resend code'}
        </button>
      </div>
    </div>
  );
}

/**
 * Landing page for the password-reset link we email hosts.
 *
 * The email contains: https://bhookme.com/host-reset-password?token=<uuid>
 * When the user clicks it, this page loads, reads the token from the
 * query string, shows a new-password form, and POSTs to
 * /api/auth/host-reset-password. On success:
 *   - the backend returns a fresh session (token + user info)
 *   - we save it into AuthContext
 *   - we start the host session
 *   - we show "Password changed successfully" for 2 seconds
 *   - we navigate to /host/dashboard
 */
export default function HostResetPasswordPage() {
  const location = useLocation();
  const navigate = useNavigate();
  const { setSessionFromResponse, startHostSession } = useAuth();

  const [token, setToken] = useState('');
  const [password, setPassword] = useState('');
  const [confirm, setConfirm] = useState('');
  const [error, setError] = useState('');
  const [submitting, setSubmitting] = useState(false);
  const [success, setSuccess] = useState(false);

  useEffect(() => {
    const params = new URLSearchParams(location.search);
    const t = params.get('token') || '';
    setToken(t);
    if (!t) {
      setError('This reset link is missing its token. Please request a new one.');
    }
  }, [location.search]);

  const handleSubmit = async (e) => {
    e.preventDefault();
    setError('');

    if (password.length < 6) {
      setError('Password must be at least 6 characters.');
      return;
    }
    if (password !== confirm) {
      setError('Passwords do not match.');
      return;
    }

    setSubmitting(true);
    try {
      const res = await fetch(`${API_BASE_URL}/auth/host-reset-password`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ token, newPassword: password }),
      });
      const data = await res.json();

      if (!res.ok || !data.success) {
        setError(data.message || 'Could not reset your password. Please try again.');
        return;
      }

      // Auto-login: save the session and mark as host-verified.
      setSessionFromResponse(data);
      startHostSession(data.email);

      setSuccess(true);

      // Give the user a beat to see the success message, then go.
      setTimeout(() => {
        navigate('/host/dashboard', { replace: true });
      }, 2000);
    } catch (err) {
      setError('Network error. Please try again.');
    } finally {
      setSubmitting(false);
    }

function formatDate(dateString) {
  const [y, m, d] = dateString.split('-').map(Number);
  return new Date(y, m - 1, d).toLocaleDateString('en-US', { month: 'short', day: 'numeric', year: 'numeric' });
}

function todayString() {
  const now = new Date();
  return `${now.getFullYear()}-${String(now.getMonth() + 1).padStart(2, '0')}-${String(now.getDate()).padStart(2, '0')}`;

     * or { "paid": false }. When paid AND status is CONFIRMED, the
     * booking blocks its dates (greyed/struck-through on the guest date
     * picker). Toggling paid back to false releases the dates.
     */
    @PutMapping("/{id}/paid")
    public ResponseEntity<?> setPaid(Authentication authentication,
                                      @PathVariable("id") Long id,
                                      @RequestBody Map<String, Object> body) {
        try {
            Boolean paid = body.get("paid") instanceof Boolean ? (Boolean) body.get("paid") : null;
            if (paid == null) {
                Map<String, String> err = new HashMap<>();
                err.put("message", "Body must include a boolean 'paid' field.");
                return ResponseEntity.badRequest().body(err);
            }
            CarBookingResponse response = carBookingService.setBookingPaid(authentication.getName(), id, paid);
            return ResponseEntity.ok(response);
        } catch (RuntimeException e) {
            return error(e);
        }

package bhookme_backend.security;

import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.http.HttpMethod;
import org.springframework.security.config.Customizer;
import org.springframework.security.config.annotation.web.builders.HttpSecurity;
import org.springframework.security.config.annotation.web.configuration.EnableWebSecurity;
import org.springframework.security.config.http.SessionCreationPolicy;
import org.springframework.security.crypto.bcrypt.BCryptPasswordEncoder;
import org.springframework.security.crypto.password.PasswordEncoder;
import org.springframework.security.web.SecurityFilterChain;
import org.springframework.security.web.authentication.UsernamePasswordAuthenticationFilter;

@Configuration
@EnableWebSecurity
public class SecurityConfig {

    private final JwtAuthenticationFilter jwtAuthenticationFilter;

    public SecurityConfig(JwtAuthenticationFilter jwtAuthenticationFilter) {
        this.jwtAuthenticationFilter = jwtAuthenticationFilter;
    }

    @Bean
    public PasswordEncoder passwordEncoder() {
        return new BCryptPasswordEncoder();
    }

    @Bean
    public SecurityFilterChain securityFilterChain(HttpSecurity http) throws Exception {
        http
            .cors(Customizer.withDefaults())
            .csrf(csrf -> csrf.disable())
            .sessionManagement(session -> session.sessionCreationPolicy(SessionCreationPolicy.STATELESS))
            .authorizeHttpRequests(auth -> auth
                .requestMatchers(HttpMethod.OPTIONS, "/**").permitAll()
                // Must come before the blanket /api/auth/** permitAll below -
                // Spring Security matches rules in order, so this narrower,
                // authenticated-only rule has to be listed first or the
                // broader permitAll would win and let anyone call it with
                // no login at all.
                .requestMatchers(HttpMethod.PUT, "/api/auth/set-password").authenticated()
                .requestMatchers("/api/auth/**").permitAll()
                .requestMatchers(HttpMethod.POST, "/api/admin-auth/login").permitAll()
                // Must be permitAll here too, even though it's not truly
                // public - the real access control (bootstrap mode vs
                // HR-only) is enforced inside AdminAuthService.createAdmin
                // itself, since Spring Security has no way to express
                // "public only if a table is empty." If this stayed
                // authenticated() here, an unauthenticated bootstrap
                // request would be rejected with a bare 403 before ever
                // reaching that service-layer check.
                .requestMatchers(HttpMethod.POST, "/api/admin-auth/create").permitAll()
                .requestMatchers(HttpMethod.GET, "/api/properties/mine").authenticated()
                .requestMatchers(HttpMethod.GET, "/api/properties/**").permitAll()
                .requestMatchers(HttpMethod.GET, "/api/bookings/quote").permitAll()
                // A host's own cars need login (listed first so the public rule below doesn't swallow it).
                .requestMatchers(HttpMethod.GET, "/api/cars/mine").authenticated()
                .requestMatchers(HttpMethod.GET, "/api/cars/**").permitAll()
                .requestMatchers(HttpMethod.GET, "/api/car-bookings/quote").permitAll()
                // Experiences: a host's own list needs login (listed first so the
                // public rule below doesn't swallow it); browsing and detail pages
                // are open to everyone. Creating/editing still requires login via
                // anyRequest().authenticated() at the bottom.
                .requestMatchers(HttpMethod.GET, "/api/experiences/mine").authenticated()
                .requestMatchers(HttpMethod.GET, "/api/experiences", "/api/experiences/**").permitAll()
                // Footer: newsletter signup and contact form are open to everyone.
                .requestMatchers(HttpMethod.POST, "/api/newsletter/subscribe", "/api/contact").permitAll()
                // Anyone with no account yet can apply to become a host.
                .requestMatchers(HttpMethod.POST, "/api/host-applications/public").permitAll()
                // Anyone can read a listing's approved reviews before logging in.
                .requestMatchers(HttpMethod.GET, "/api/reviews/property/**").permitAll()
                .requestMatchers(HttpMethod.GET, "/api/reviews/car/**").permitAll()
                .anyRequest().authenticated()
            )
            .addFilterBefore(jwtAuthenticationFilter, UsernamePasswordAuthenticationFilter.class);

        return http.build();
    }
