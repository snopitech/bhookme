function nightsBetween(checkIn, checkOut) {
  if (!checkIn || !checkOut) return 0;
  const [y1, m1, d1] = checkIn.split('-').map(Number);
  const [y2, m2, d2] = checkOut.split('-').map(Number);
  const diff = Math.round((Date.UTC(y2, m2 - 1, d2) - Date.UTC(y1, m1 - 1, d1)) / 86400000);
  return diff > 0 ? diff : 0;
}

function shortDate(iso) {
  const [y, m, d] = iso.split('-').map(Number);
  return new Date(y, m - 1, d).toLocaleDateString(undefined, { month: 'short', day: 'numeric' });
}

function money(n) {
  return `$${Math.round(n).toLocaleString()}`;
}

  const price = priceOf(listing);

  return (
    <Link to={`/property/${listing.id}${query ? `?${query}` : ''}`} className="group block">
      <div className="aspect-square rounded-xl overflow-hidden bg-gray-200">
        {img ? (
          <img
            src={img}
            alt={title}
            loading="lazy"
            className="w-full h-full object-cover group-hover:scale-105 transition duration-300"
          />
        ) : (
          <div className="w-full h-full flex items-center justify-center text-4xl" aria-hidden="true">🏠</div>

      </div>
      <div className="mt-3">
        <p className="font-semibold truncate">{place || title}</p>
        {place && <p className="text-sm text-gray-500 truncate">{title}</p>}
        {price ? (
          <p className="mt-1 text-sm">

          </p>
        ) : (

// yyyy-mm-dd in the visitor's own time zone
function localISO(date) {
  const d = new Date(date.getTime() - date.getTimezoneOffset() * 60000);
  return d.toISOString().slice(0, 10);
}

function addDays(iso, days) {
  const [y, m, d] = iso.split('-').map(Number);
  return localISO(new Date(y, m - 1, d + days));
}

function shortDate(iso) {
  if (!iso) return '';
  const [y, m, d] = iso.split('-').map(Number);
  return new Date(y, m - 1, d).toLocaleDateString(undefined, { month: 'short', day: 'numeric' });
}

function Stepper({ label, hint, value, min, onChange }) {
  return (
    <div className="flex items-center justify-between py-4 border-b border-gray-100 last:border-b-0">
      <div>
        <p className="font-semibold text-sm">{label}</p>
        <p className="text-xs text-gray-500">{hint}</p>
      </div>
      <div className="flex items-center gap-3">
        <button
          type="button"
          onClick={() => onChange(Math.max(min, value - 1))}
          disabled={value <= min}
          aria-label={`Fewer ${label.toLowerCase()}`}
          className="w-8 h-8 rounded-full border border-gray-300 text-gray-600 hover:border-gray-900 disabled:opacity-30 disabled:hover:border-gray-300"
        >
          −
        </button>
        <span className="w-5 text-center text-sm">{value}</span>
        <button
          type="button"
          onClick={() => onChange(Math.min(16, value + 1))}
          aria-label={`More ${label.toLowerCase()}`}
          className="w-8 h-8 rounded-full border border-gray-300 text-gray-600 hover:border-gray-900"
        >
          +
        </button>
      </div>
    </div>
  );
}

  useEffect(() => {

    const p = new URLSearchParams(routerLocation.search);
    setServiceType(p.get('type') || '');
    if (path.startsWith('/category/experiences')) {
      setWhere(p.get('location') || '');
      setCheckIn(p.get('date') || '');
      setCheckOut('');

// Round photo picker for the reviewer's avatar (shown where the initial
// letter would otherwise be). Uploads one image to /api/upload/image.
function AvatarPicker({ value, onChange, name, token, onUploadingChange }) {
  const inputRef = React.useRef(null);
  const [uploading, setUploading] = useState(false);
  const [error, setError] = useState('');
  const initial = (name || '?').trim().charAt(0).toUpperCase() || '?';

  const pick = async (file) => {
    if (!file) return;
    setError('');
    if (!file.type.startsWith('image/')) {
      setError('Please choose an image.');
      return;
    }
    if (file.size > 10 * 1024 * 1024) {
      setError('Photo must be under 10MB.');
      return;
    }
    setUploading(true);
    onUploadingChange?.(true);
    try {
      const form = new FormData();
      form.append('file', file);
      form.append('folder', 'profiles');
      const res = await fetch(`${API_BASE_URL}/upload/image`, {
        method: 'POST',
        headers: { Authorization: `Bearer ${token}` },
        body: form,
      });
      const data = await res.json().catch(() => null);
      if (!res.ok || !data?.url) throw new Error((data && data.message) || 'Upload failed.');
      onChange(data.url);
    } catch (err) {
      setError(err.message || 'Upload failed.');
    } finally {
      setUploading(false);
      onUploadingChange?.(false);
    }
  };

  return (
    <div className="flex items-center gap-3">
      <button
        type="button"
        onClick={() => inputRef.current?.click()}
        disabled={uploading}
        aria-label={value ? 'Change your photo' : 'Add your photo'}
        className="relative w-14 h-14 rounded-full overflow-hidden bg-gray-200 text-gray-500 flex items-center justify-center text-lg font-semibold border border-gray-300 hover:opacity-90 shrink-0"
      >
        {value ? <img src={value} alt="Your photo" className="w-full h-full object-cover" /> : initial}
        {uploading && (
          <span className="absolute inset-0 bg-white/70 flex items-center justify-center text-xs text-gray-700">...</span>
        )}
      </button>
      <div className="text-sm">
        <button type="button" onClick={() => inputRef.current?.click()} disabled={uploading} className="font-medium text-gray-800 underline">
          {uploading ? 'Uploading...' : value ? 'Change photo' : 'Add your photo'}
        </button>
        {value && !uploading && (
          <button type="button" onClick={() => onChange(null)} className="ml-3 text-gray-500 underline">
            Remove
          </button>
        )}
        <p className="text-xs text-gray-400">Optional. Shown next to your name.</p>
        {error && <p className="text-xs text-red-500">{error}</p>}
      </div>
      <input
        ref={inputRef}
        type="file"
        accept="image/jpeg,image/png,image/webp"
        className="hidden"
        onChange={(e) => {
          pick(e.target.files?.[0]);
          e.target.value = '';
        }}
      />
    </div>
  );
}

    @GetMapping

                                    @RequestParam(required = false) String checkIn,
                                    @RequestParam(required = false) String checkOut,
                                    @RequestParam(required = false) Integer guests) {
        try {

    // Keeps rows where every word appears in at least one of searchKeys,
    // ranked by how many words appear in the titleKeys.
    private List<Map<String, Object>> rank(List<Map<String, Object>> rows, List<String> words,
                                           String[] titleKeys, String[] searchKeys) {
        List<Map<String, Object>> matched = new ArrayList<>();
        Map<Map<String, Object>, Integer> scores = new java.util.IdentityHashMap<>();
        for (Map<String, Object> row : rows) {
            String hay = joinValues(row, searchKeys);
            boolean all = true;
            for (String w : words) {
                if (!hay.contains(w)) { all = false; break; }
            }
            if (!all) continue;
            String title = joinValues(row, titleKeys);
            int score = 0;
            for (String w : words) {
                if (title.contains(w)) score++;
            }
            scores.put(row, score);
            matched.add(row);
        }
        matched.sort((a, b) -> Integer.compare(scores.get(b), scores.get(a)));
        return matched;
    }

    private String joinValues(Map<String, Object> row, String[] keys) {
        StringBuilder sb = new StringBuilder();
        for (String k : keys) {
            Object v = row.get(k);
            if (v != null) sb.append(' ').append(v.toString().toLowerCase());
        }
        return sb.toString();
    }

    private boolean matchesLocation(Map<String, Object> row, List<String> tokens) {
        StringBuilder hay = new StringBuilder();
        for (Map.Entry<String, Object> e : row.entrySet()) {
            if (e.getValue() != null && LOCATION_KEY.matcher(e.getKey()).matches()) {
                hay.append(' ').append(e.getValue().toString().toLowerCase());
            }
        }
        Object title = row.get("title");
        if (title != null) hay.append(' ').append(title.toString().toLowerCase());
        String haystack = hay.toString();
        for (String t : tokens) {
            if (!haystack.contains(t)) return false;
        }
        return true;
    }

    private Integer capacityOf(Map<String, Object> row) {
        for (Map.Entry<String, Object> e : row.entrySet()) {
            if (CAPACITY_KEY.matcher(e.getKey()).matches() && e.getValue() instanceof Number) {
                return ((Number) e.getValue()).intValue();
            }
        }
        return null;
    }

    private LocalDate parseDate(String value) {
        if (value == null || value.isBlank()) return null;
        return LocalDate.parse(value.trim());

    private String text(Object value) {
        if (value == null) return null;
        String s = value.toString().trim();
        return s.isEmpty() ? null : s;
    }

    private String firstText(Map<String, Object> row, String... keys) {
        for (String k : keys) {
            String v = text(row.get(k));
            if (v != null) return v;

        return null;

    private String rootMessage(RuntimeException e) {
        Throwable root = e;
        while (root.getCause() != null && root.getCause() != root) root = root.getCause();
        return root.getMessage() != null ? root.getMessage() : e.getMessage();

    List<Property> findByHostId(Long hostId);

    // ============================================================
    // NEAREST PROPERTIES (Haversine formula, in miles)
    // ------------------------------------------------------------
    // Powers the homepage's first section: "properties closest to
    // the searching user's location". Only considers ACTIVE
    // properties.
    // ============================================================
    @Query(value =
        "SELECT p.*, " +
        "(3959 * ACOS(COS(RADIANS(:lat)) * COS(RADIANS(p.latitude)) * " +
        "COS(RADIANS(p.longitude) - RADIANS(:lng)) + " +
        "SIN(RADIANS(:lat)) * SIN(RADIANS(p.latitude)))) AS distance_miles " +
        "FROM properties p " +
        "WHERE p.status = 'ACTIVE' " +
        "HAVING distance_miles IS NOT NULL " +
        "ORDER BY distance_miles ASC",
        countQuery = "SELECT COUNT(*) FROM properties WHERE status = 'ACTIVE'",
        nativeQuery = true)
    List<Property> findNearestActiveProperties(@Param("lat") double latitude,
                                                @Param("lng") double longitude,
                                                Pageable pageable);
